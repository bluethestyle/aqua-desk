# 02. 데이터 모델

> v2 (서버권위·RLS 재작성 반영)

> **대원칙 — 서버 권위.** 경제·수집·결제·토큰 테이블은 **클라이언트 직접 쓰기 금지**. 모든 증감은 `SECURITY DEFINER` 함수 또는 service-role Edge Function으로만 수행한다. 클라이언트는 **자기 행 SELECT만** 한다.

## 1. Postgres 스키마 (Supabase)

> `text` 카탈로그 ID(`species_id`, `item_id`, `theme_id`, `spot_id`)는 에셋 키와 1:1. 사용자 데이터는 모두 `user_id` 스코프.

### 1.1 카탈로그 (public read)

```sql
-- 테마 카탈로그 (그래픽 + 네이티브 셰이더 무드)
create table themes (
  id            text primary key,                 -- 'deepsea' | 'cyberpunk' | 'emerald' ...
  name          text not null,
  price_pearl   int,                              -- nullable(기본 무료 테마면 null)
  asset_prefix  text not null,                    -- storage 경로 prefix (sprites/theme/{id})
  shader_profile text not null,                   -- 네이티브 FX 프로파일 키 (LUT + FX 패스)
  is_premium    boolean default false
);

-- 어종 카탈로그 (시드)
create table fish_species (
  id          text primary key,
  rarity      text not null,                      -- common | rare | mythic
  body_type   text not null,                      -- fusiform | disc | eel
  base_sprite text not null,
  is_hero     boolean default false,              -- 테마 전용 도트
  theme_id    text references themes(id)          -- 테마 시그니처(nullable)
);

-- 아이템 카탈로그 (장식/먹이/테마)
create table items (
  id          text primary key,
  type        text not null,                      -- deco | food | theme
  price_coin  int,
  price_pearl int,                                -- nullable(유료 전용이면 set)
  asset_key   text not null,
  theme_id    text references themes(id)          -- 테마 스킨이면 set
);

-- 낚시터 카탈로그 (낚시 풀 정의)
create table fishing_spots (
  id             text primary key,                -- 'home' ...
  name           text not null,
  species_pool   jsonb not null,                  -- ["clownfish","tang",...] 후보 종
  rarity_weights jsonb not null                   -- {"common":0.7,"rare":0.25,"mythic":0.05}
);

-- 시너지 카탈로그 (물고기-장식 조합 → 버프)
create table synergies (
  id              text primary key,
  fish_species_id text references fish_species(id),
  deco_item_id    text references items(id),
  bonus           jsonb not null                  -- {"coin_rate_bonus":0.02, ...}
);
```

> **시드**: `themes`는 `deepsea`(무료 기본), `cyberpunk`, `emerald` 3종. `fishing_spots`는 `home` 1곳.

### 1.2 사용자 데이터

```sql
-- 사용자 프로필 (auth.users 1:1)
create table profiles (
  id            uuid primary key references auth.users(id) on delete cascade,
  display_name  text,
  aqua_pass_until timestamptz,                    -- 구독 만료(null=미구독) — 서버 전용
  settings      jsonb default '{}',               -- {dayNight, lowPower, sound...}
  created_at    timestamptz default now()
);

-- 지갑 (재화) — 권위 테이블, 클라 SELECT-only
create table wallets (
  user_id            uuid primary key references profiles(id) on delete cascade,
  coins              bigint default 0,
  pearls             int    default 0,
  stamina            int    default 5,
  stamina_updated_at timestamptz default now(),
  hearts             int    default 0,            -- 소셜 카운터(임계 도달 시 코인 환산)
  coin_rate_bonus    real   default 0             -- 도감 마일스톤/시너지 버프 누적(서버 갱신)
);

-- 어항 (멀티 어항: index 1,2,...) — owner write(제한)
create table aquariums (
  id              uuid primary key default gen_random_uuid(),
  user_id         uuid not null references profiles(id) on delete cascade,
  idx             int  not null default 1,        -- 어항 번호
  theme_id        text not null default 'deepsea' references themes(id),
  slots           jsonb default '[]',             -- [{slotId,itemId,x,y,layer}] — 클라 UPDATE 허용
  water_quality   real default 1.0,               -- 0..1 (청소로 회복) — 서버 전용
  version         int  default 1,                 -- 서버 전용(CAS)
  rating          int  default 0,                 -- 풍수/감성 점수, 서버 계산
  last_collected_at timestamptz default now(),    -- 오프라인 적립 기준
  updated_at      timestamptz default now(),
  unique (user_id, idx)
);

-- 물고기 인스턴스 (user_id 없음 → 어항 소유로 권한 유도)
create table fish_instances (
  id            uuid primary key default gen_random_uuid(),
  aquarium_id   uuid references aquariums(id) on delete cascade,
  species_id    text references fish_species(id),
  nickname      text,
  nature        text not null,                    -- timid|gluttonous|curious|lone
  growth_stage  int  default 1,                   -- 1..4
  hunger        real default 1.0,                 -- 0..1 (0=불만족 진입)
  size_pct      real default 0.5,                 -- 0..1, 낚시 시 서버 산출(스킬 보정)
  last_fed_at   timestamptz default now(),
  born_at       timestamptz default now()
);
-- 불만족 상태는 satisfied=false 파생값(hunger==0 또는 water_quality 저하)으로 계산 → 별도 컬럼 불필요.

-- 인벤토리 — 권위 테이블, 클라 SELECT-only
create table inventory (
  user_id uuid references profiles(id) on delete cascade,
  item_id text references items(id),
  qty     int default 0,
  primary key (user_id, item_id)
);

-- 도감 (수집 메타의 척추) — 권위 테이블, 클라 SELECT-only
create table dex_entries (
  user_id        uuid references profiles(id) on delete cascade,
  species_id     text references fish_species(id),
  first_caught_at timestamptz default now(),
  count          int default 1,
  primary key (user_id, species_id)
);

-- 낚시 세션 (멱등·스태미나 게이트·중복지급 방지 키)
create table fishing_sessions (
  id                      uuid primary key default gen_random_uuid(),
  user_id                 uuid references profiles(id) on delete cascade,
  spot_id                 text references fishing_spots(id),
  started_at              timestamptz default now(),
  consumed                boolean default false,         -- resolve 완료 표식(멱등)
  reward_fish_instance_id uuid references fish_instances(id)
);

-- 공유 토큰 (★불투명, user_id 비노출 R6) — RPC 전용(직접 정책 없음)
create table share_tokens (
  token        text primary key,                  -- 랜덤 16자+ 불투명
  user_id      uuid references profiles(id) on delete cascade,
  aquarium_idx int default 1,
  created_at   timestamptz default now(),
  expires_at   timestamptz
);

-- 소셜 선물(하트) — 지연 전달, 권위 테이블
create table gifts (
  id           uuid primary key default gen_random_uuid(),
  to_user_id   uuid references profiles(id) on delete cascade,
  from_user_id uuid references profiles(id),
  type         text default 'heart',
  qty          int default 1,
  claimed      boolean default false,
  created_at   timestamptz default now()
);

-- 결제 로그 — 권위 테이블(영수증 멱등)
create table purchases (
  id         uuid primary key default gen_random_uuid(),
  user_id    uuid references profiles(id) on delete cascade,
  product_id text not null,
  platform   text not null,                       -- play | appstore
  receipt    text,
  verified   boolean default false,
  created_at timestamptz default now(),
  unique (platform, receipt)                      -- 영수증 멱등(중복 지급 방지)
);
```

> **소셜 관계 모델**: `friends` 테이블은 없다. 소셜은 **토큰 기반**이며(`share_tokens`), '내 share token을 받은 누구나' 하트를 보낼 수 있다(레이트리밋 + 자기선물 차단). 자세한 흐름은 [04 §5](./04-웹앱-백엔드-설계.md)·아키텍처 문서 참조.

## 2. RLS 정책 (서버 권위)

RLS는 **SELECT 권한**과 **WRITE 권한**을 분리한다. 권위 테이블은 클라이언트가 **자기 행 SELECT만** 가능하고, 쓰기 정책은 **부여하지 않는다**(= 클라 쓰기 거부). 모든 쓰기는 `SECURITY DEFINER` 함수 / service-role Edge Function이 RLS를 우회해 수행한다.

### 2.1 권한 분류 표

| 테이블 | SELECT 권한 | WRITE 권한 |
|--------|-------------|------------|
| `wallets` | owner (`auth.uid()=user_id`) | **서버 전용** (쓰기 정책 없음) |
| `inventory` | owner (`auth.uid()=user_id`) | **서버 전용** |
| `dex_entries` | owner (`auth.uid()=user_id`) | **서버 전용** |
| `purchases` | owner (`auth.uid()=user_id`) | **서버 전용** |
| `gifts` | owner (`auth.uid()=to_user_id`) | **서버 전용** (`send_heart`/`claim_gifts`) |
| `fish_instances` | 어항 소유자 (EXISTS join) | **서버 전용** (쓰기 정책 없음) |
| `fishing_sessions` | owner (`auth.uid()=user_id`) | **서버 전용** (`start_fishing`/`fishing-resolve`) |
| `share_tokens` | **없음(거부)** | **없음(거부)** — `issue_share_token`/`get_shared_aquarium` RPC 전용 |
| `aquariums` | owner (`auth.uid()=user_id`) | owner UPDATE는 **`slots`만**. `theme_id`는 `set_aquarium_theme` RPC, `water_quality`/`version`/`rating`은 서버 전용 |
| `profiles` | owner (`auth.uid()=id`) | owner UPDATE는 **`display_name`·`settings`만**. `aqua_pass_until`은 서버 전용(BEFORE UPDATE 트리거로 보호) |
| `themes`, `items`, `fish_species`, `fishing_spots`, `synergies` | **public read** (카탈로그) | 서버/관리자 전용 |

### 2.2 정책 SQL

```sql
-- ── 카탈로그: public read ──────────────────────────────
alter table themes        enable row level security;
alter table items         enable row level security;
alter table fish_species  enable row level security;
alter table fishing_spots enable row level security;
alter table synergies     enable row level security;

create policy themes_read        on themes        for select using (true);
create policy items_read         on items         for select using (true);
create policy fish_species_read  on fish_species  for select using (true);
create policy fishing_spots_read on fishing_spots for select using (true);
create policy synergies_read     on synergies     for select using (true);

-- ── 권위 테이블: 자기 행 SELECT만 (쓰기 정책 없음 = 클라 거부) ──
alter table wallets     enable row level security;
alter table inventory   enable row level security;
alter table dex_entries enable row level security;
alter table purchases   enable row level security;

create policy wallets_read     on wallets     for select using (auth.uid() = user_id);
create policy inventory_read   on inventory   for select using (auth.uid() = user_id);
create policy dex_read         on dex_entries for select using (auth.uid() = user_id);
create policy purchases_read   on purchases   for select using (auth.uid() = user_id);
-- (INSERT/UPDATE/DELETE 정책 미부여 = 클라 쓰기 거부. 쓰기는 SECURITY DEFINER 함수가 수행.)

-- gifts: 받은 선물만 SELECT (쓰기 정책 없음 → send_heart/claim_gifts RPC)
alter table gifts enable row level security;
create policy gifts_read on gifts for select using (auth.uid() = to_user_id);

-- fishing_sessions: 자기 세션 SELECT만 (쓰기 정책 없음 → start_fishing/fishing-resolve)
alter table fishing_sessions enable row level security;
create policy fishing_sessions_read on fishing_sessions for select using (auth.uid() = user_id);

-- ── fish_instances: user_id 없음 → 어항 소유 EXISTS join으로 SELECT ──
alter table fish_instances enable row level security;
create policy fish_read on fish_instances for select using (
  exists (
    select 1 from aquariums a
    where a.id = fish_instances.aquarium_id
      and a.user_id = auth.uid()
  )
);
-- (INSERT/UPDATE/DELETE 정책 미부여 = 클라 거부. 쓰기는 SECURITY DEFINER 함수가 수행.)

-- ── share_tokens: 정책 없음 = 모든 직접 SELECT/INSERT 거부 ──
alter table share_tokens enable row level security;
-- 정책을 부여하지 않는다. issue_share_token / get_shared_aquarium RPC(SECURITY DEFINER)로만 접근.

-- ── aquariums: 소유자 SELECT + slots만 클라 UPDATE ──
alter table aquariums enable row level security;
create policy aquariums_read on aquariums
  for select using (auth.uid() = user_id);
create policy aquariums_update_slots on aquariums
  for update using (auth.uid() = user_id) with check (auth.uid() = user_id);
-- theme_id/water_quality/version/rating 변경은 컬럼 단위 트리거로 보호(아래 §3.2).

-- ── profiles: 소유자 SELECT + display_name/settings만 클라 UPDATE ──
alter table profiles enable row level security;
create policy profiles_read on profiles
  for select using (auth.uid() = id);
create policy profiles_update_self on profiles
  for update using (auth.uid() = id) with check (auth.uid() = id);
-- aqua_pass_until 변경은 BEFORE UPDATE 트리거로 차단(아래 §3.2).
```

> 권위 테이블의 쓰기 정책이 **존재하지 않으므로** anon 키 클라이언트의 모든 INSERT/UPDATE/DELETE는 RLS에서 거부된다. 합법적 쓰기는 모두 [§4](#4-서버-권위-함수rpc) 함수가 담당한다.

## 3. 부트스트랩 / 보호 트리거

### 3.1 신규 사용자 부트스트랩

회원 가입(`auth.users` insert) 시 `profiles`·`wallets` 행을 자동 생성한다.

```sql
create function handle_new_user() returns trigger
language plpgsql security definer as $$
begin
  insert into profiles(id) values (new.id);
  insert into wallets(user_id) values (new.id);
  return new;
end;
$$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function handle_new_user();
```

### 3.2 컬럼 보호 트리거 (서버 전용 필드 방어)

owner UPDATE 정책이 행 단위만 통제하므로, 서버 전용 컬럼은 BEFORE UPDATE 트리거로 변경을 무시(원값 복원)한다.

```sql
-- 세션 권위 플래그: 보호 컬럼을 쓰는 SECURITY DEFINER RPC가 쓰기 직전 설정(트랜잭션 한정).
-- service-role Edge Function은 auth.role()='service_role'이라 플래그 없이 통과.
create function app_is_server_write() returns boolean
language sql stable as $$
  select coalesce(current_setting('app.authority_write', true), 'off') = 'on'
      or auth.role() = 'service_role';
$$;

-- profiles: aqua_pass_until은 서버만 변경 가능
create function guard_profiles() returns trigger
language plpgsql security definer as $$
begin
  if not app_is_server_write() then
    new.aqua_pass_until := old.aqua_pass_until;   -- 보호 필드 복원
  end if;
  return new;
end;
$$;
create trigger profiles_guard before update on profiles
  for each row execute function guard_profiles();

-- aquariums: theme_id/water_quality/version/rating/last_collected_at는 서버만 변경
create function guard_aquariums() returns trigger
language plpgsql security definer as $$
begin
  if not app_is_server_write() then
    new.theme_id          := old.theme_id;        -- set_aquarium_theme RPC로만
    new.water_quality     := old.water_quality;
    new.version           := old.version;
    new.rating            := old.rating;
    new.last_collected_at := old.last_collected_at;
  end if;
  return new;
end;
$$;
create trigger aquariums_guard before update on aquariums
  for each row execute function guard_aquariums();
```

> 클라이언트는 `aquariums.slots`와 `profiles.{display_name,settings}`만 실효 변경 가능. 그 외 컬럼은 트리거가 원값으로 되돌린다.
>
> ⚠️ **중요 — `SECURITY DEFINER` ≠ `service_role`**: `SECURITY DEFINER`는 함수를 *정의자(postgres) 권한*으로 실행해 RLS는 우회하지만, `auth.role()`은 **호출자 JWT의 role(`authenticated`)을 그대로 반환**한다(컨텍스트를 `service_role`로 바꾸지 않음). 따라서 보호 컬럼을 변경하는 SECURITY DEFINER RPC(`feed_fish`/`clean_aquarium`/`collect_offline`/`set_aquarium_theme` 등)는 보호 컬럼 쓰기 **직전에** `perform set_config('app.authority_write','on', true);`(트랜잭션 한정 플래그)를 호출해야 트리거를 통과한다. service-role Edge Function(`verify-receipt` 등)은 이미 `auth.role()='service_role'`이라 플래그가 불필요하다.

## 4. 서버 권위 함수/RPC

모든 상태 변경은 `SECURITY DEFINER` 함수 또는 service-role Edge Function이 수행한다. **동시성**: 모든 상태변경 RPC는 `version` CAS(`UPDATE ... WHERE id=? AND version=?`; 불일치 시 `conflict` 반환 → 클라 refresh). last-write-wins를 대체한다. 상세 책임은 [04 §5](./04-웹앱-백엔드-설계.md) 참조.

### 4.1 함수 시그니처 (02 권위)

| 함수 / RPC | 구현 | 시그니처 | 책임 요약 |
|------------|------|----------|-----------|
| `start_fishing` | SECURITY DEFINER RPC | `start_fishing(spot_id text) → uuid` | lazy 스태미나 재생(`stamina_updated_at` 기준 +1/N분, cap) → `stamina-1` → `fishing_sessions` insert → `session_id` 반환(부족 시 거부) |
| `fishing-resolve` | Edge / service role | `fishing-resolve(session_id uuid, skill jsonb)` | session 미consumed 검증 → `species_pool`+`rarity_weights`+seed로 종 결정, skill로 `size_pct`·희귀확률 보정 → `fish_instances` insert + `dex` upsert + `session.consumed=true` + reward 연결. **멱등**(이미 consumed면 기존 reward 반환) |
| `feed_fish` | SECURITY DEFINER RPC | `feed_fish(aquarium_id uuid)` | `hunger` 갱신 + `version` CAS |
| `clean_aquarium` | SECURITY DEFINER RPC | `clean_aquarium(aquarium_id uuid)` | `water_quality` 갱신 + `version` CAS |
| `collect_offline` | SECURITY DEFINER RPC | `collect_offline(aquarium_id uuid)` | `last_collected_at`·`aqua_pass_until` 기준 코인 적립(캡 6h, 패스 24h) + `last_collected_at` 갱신 + `coin_rate_bonus` 반영 |
| `purchase_item` | SECURITY DEFINER RPC | `purchase_item(item_id text)` | 가격(coin/pearl) 원자 차감 + `inventory` upsert |
| `set_aquarium_theme` | SECURITY DEFINER RPC | `set_aquarium_theme(aquarium_id uuid, theme_id text)` | 소유 검증 후 `theme_id` 변경(프리미엄 보유 확인) |
| `daily-shop-roll` | Edge | `daily-shop-roll()` | `date + user` seed 6슬롯 |
| `verify-receipt` | Edge / service role | `verify-receipt(platform text, receipt text)` | 영수증 검증, `purchases.receipt` UNIQUE로 멱등, pearls/items 원자 지급, `verified=true` |
| `issue_share_token` | SECURITY DEFINER RPC | `issue_share_token(aquarium_idx int) → text` | 16자+ 불투명 토큰 생성·저장, **토큰만 반환** |
| `get_shared_aquarium` | SECURITY DEFINER RPC | `get_shared_aquarium(token text) → jsonb` | 만료 검사 후 read-only 스냅샷 반환, **`user_id` 절대 미반환** |
| `send_heart` | SECURITY DEFINER RPC | `send_heart(to_token text)` | 토큰→대상 user 해석, 자기선물 차단, 레이트리밋(대상별 일 5회), `gifts` insert |
| `claim_gifts` | SECURITY DEFINER RPC | `claim_gifts()` | 미수령 `gifts`→`wallets.hearts` 적립 + `claimed=true`. 하트→코인 자동환산: `hearts` 임계(100) 도달 시 claim 내에서 코인 환산(100 hearts → 1000 coins) |
| `grant_ad_reward` | Edge | `grant_ad_reward(kind text)` | 리워드광고 보상(`stamina+1` / `offline x2` / `shop refresh`) 서버 지급, 1일 횟수 제한(광고 SSV/검증) |

> **멱등 키**: 낚시 임시 지급/중복 방지의 멱등 키는 `fishing_sessions.id`. 결제 멱등 키는 `purchases (platform, receipt)` UNIQUE.

## 6. 스냅샷 계약 (Native ↔ Web 공유 JSON)

배경 렌더 전용 경량 포맷. `packages/game-spec`(순수 TS)이 DB row → 스냅샷으로 변환한다.

```typescript
interface AquariumSnapshot {
  version: number;
  themeId: string;
  dayNight: 'auto' | 'day' | 'night';
  waterQuality: number;           // 0..1
  slots: SlotPlacement[];
  fish: FishSnapshot[];
  pendingAnims: PendingAnim[];    // ★비가시 큐잉(R1): 'feed-drop'|'clean'|'heart'
}

interface SlotPlacement {
  slotId: string;
  itemId: string;
  x: number; y: number;           // 초기 배치 힌트(서버 시드) — 런타임 위치 권위는 네이티브 FSM
  layer: number;
}

interface FishSnapshot {
  id: string;
  speciesId: string;
  bodyType: 'fusiform' | 'disc' | 'eel';
  nickname?: string;
  nature: 'timid' | 'gluttonous' | 'curious' | 'lone';
  growthStage: number;            // 1..4
  sizePct: number;                // 0..1, 개체 크기(낚시 시 서버 산출)
  satisfied: boolean;             // hunger/water_quality 파생(불만족=false)
  // ★ x/y는 '초기 배치 힌트'(서버 시드)이며, 런타임 위치 권위는 네이티브 FSM이다.
  x: number; y: number;           // 0..1 정규화 좌표(초기 배치 힌트)
}

// 스프라이트 state enum: idle | swim | scatter | peek | yawn | sulk
interface PendingAnim { kind: 'feed-drop' | 'clean' | 'heart'; at: number; payload?: unknown; }
```

> **위치 권위**: `SlotPlacement.x/y`·`FishSnapshot.x/y`는 서버가 시드하는 **초기 배치 힌트**일 뿐, 매 프레임의 실제 위치 권위는 **네이티브 FSM**이 갖는다(스냅샷은 위치를 강제하지 않는다).

## 7. 브리지 / 딥링크 API

```typescript
// Web → Native (window.AquaDesk)
interface AquaDeskBridge {
  minBridgeVersion: number;                       // ★버전 협상(하위 미충족 시 폴백)

  applySnapshot(s: AquariumSnapshot): Promise<void>;     // 즉시 반영(실패 콜백 포함)
  requestSnapshot(): Promise<AquariumSnapshot>;          // 현재 캐시 조회
  previewWallpaper(): void;                              // 배경 설정 화면 유도(ACTION_CHANGE_LIVE_WALLPAPER)
  getAuthSession(): Promise<AuthSession | null>;         // ★네이티브 세션 읽기(R8, 웹 독립 refresh 금지)
  setAuthSession(refreshToken: string): Promise<void>;   // ★백그라운드 인증(R8)
  clearAuthSession(): void;
  setLowPower(on: boolean): void;                        // 셰이더 게이팅 토글
}

interface BridgeError { code: string; message: string; }  // 각 호출 실패/에러 콜백 공통

// Native → Web
//   딥링크: aquadesk://sync-complete?version=123
//           aquadesk://aquarium/{shareToken}     ← 친구 어항 뷰어(불투명 토큰)
//   postMessage: { type: 'sync-complete', version } / { type: 'bridge-error', error: BridgeError } 등
```

> **버전 협상**: 각 브리지/딥링크 호출은 `minBridgeVersion`을 협상하고, 실패·에러 콜백(`BridgeError`)을 받는다. 미충족 시 웹앱은 안전 폴백(앱 업데이트 유도/기능 비활성).
>
> **세션 단일화(R8)**: 웹앱은 resume 시 `getAuthSession()`으로 네이티브 세션을 읽어 재사용한다(웹 독립 refresh 금지). invalid refresh 시 재인증을 유도한다.
>
> **보안**: 공유 URL에는 `user_id`가 아니라 `share_tokens.token`만 노출. 뷰어는 `get_shared_aquarium(token)`로 read-only 스냅샷만 조회하며 `user_id`는 절대 반환되지 않는다.
